//relations
products.id < product_variants.product_id

attributes.id < attribute_values.attribute_id

product_variants.id - variant_attributes.variant_id

product_variants.id - variant_images.variant_id

attribute_values.id - variant_attributes.attribute_value_id

products.id < product_images.product_id

products.id - product_categories.product_id
category.id - product_categories.category_id

user.user_id < user_otp.user_id

user.user_id < orders.user_id

user.user_id < user_address.user_id


invoice.id - orders.id

orders.id < order_products.order_id
product_variants.id > order_products.product_variants_id

orders.id - payment_info.order_id
// user_cart_details relations
user.user_id - cart.user_id

cart.id - cart_products.cart_id
products.id - cart_products.product_variants_id

// wishlist relation
user.user_id - wishlist_items.user_id
products.id - wishlist_items.product_id

user.user_id - coupon_usage.user_id
coupons.id - coupon_usage.coupon_id

user.user_id - reviews.user_id
products.id - reviews.product_id
